// BuildCallAggregates — backfill of the weekly/hourly summaries (rolling-year plan) // v2 (step 3a): inbound calls with no Aircall team_name are attributed by number_id via cfg_numbers, // so missed calls on team lines (e.g. Revered, Credit) are counted against that team. // Reads ALL of raw_calls in chunks and (re)builds two small summary sheets: // agg_calls_hourly : one row per UK-local date x hour x direction x team // agg_agent_week : one row per ISO week x agent // Safe to re-run: both sheets are cleared and fully rewritten each time. // Run from Excel Online (Automate tab), NOT from Power Automate (120s limit). interface HourAgg { weekKey: string; isoYear: number; isoWeek: number; date: string; weekday: number; hour: number; direction: string; team: string; calls: number; answered: number; missed: number; missedOoh: number; missedAbandoned: number; connected10s: number; talkSecs: number; } interface AgentAgg { weekKey: string; isoYear: number; isoWeek: number; user: string; team: string; inboundAnswered: number; outboundTotal: number; outboundConnected10s: number; talkInSecs: number; talkOutSecs: number; } // Inbound attribution order: 1) Aircall team_name 2) cfg_numbers (number_id -> team) // 3) built-in list below (only used for numbers missing from cfg_numbers) 4) 'Other' const NUMBER_TEAM: { [id: string]: string } = { '286844': 'Support', '286845': 'Support', '286847': 'Support', '286848': 'Support', '192211': 'Support', '201178': 'Onboarding', '1216204': 'Risk and compliance', '326618': '2nd line (Tech support team)' }; const READ_CHUNK = 20000; const WRITE_CHUNK = 5000; function main(workbook: ExcelScript.Workbook) { const raw = workbook.getWorksheet('raw_calls'); if (!raw) { console.log('ERROR: raw_calls sheet not found'); return; } // Lookups: agent -> team, number_id -> team const agentTeam = readLookup(workbook, 'cfg_agents', 'agent_name', 'team', false); const numberTeam = readLookup(workbook, 'cfg_numbers', 'number_id', 'team', true); console.log('cfg_agents entries: ' + Object.keys(agentTeam).length + ' | cfg_numbers entries: ' + Object.keys(numberTeam).length); const attrib: { [how: string]: number } = { team_name: 0, cfg_numbers: 0, built_in_list: 0, unattributed: 0 }; const inboundTeams: { [team: string]: { calls: number; answered: number; viaNumber: number } } = {}; const used = raw.getUsedRange(); const rowCount = used.getRowCount(); const colCount = used.getColumnCount(); const hdr = raw.getRangeByIndexes(0, 0, 1, colCount).getValues()[0].map(h => String(h).trim()); const col = (name: string): number => { const i = hdr.indexOf(name); if (i < 0) throw new Error('Column not found in raw_calls: ' + name); return i; }; const iId = col('id'), iDir = col('direction'), iReason = col('missed_call_reason'), iSt = col('started_at'), iAns = col('answered_at'), iEnd = col('ended_at'), iDur = col('duration'), iNum = col('number_id'), iUser = col('user_name'), iTeam = col('team_name'); const hourly: { [k: string]: HourAgg } = {}; const agents: { [k: string]: AgentAgg } = {}; const seen = new Set(); const weekCounts: { [k: string]: number } = {}; let read = 0, dupes = 0, noTs = 0; for (let start = 1; start < rowCount; start += READ_CHUNK) { const n = Math.min(READ_CHUNK, rowCount - start); const rows = raw.getRangeByIndexes(start, 0, n, colCount).getValues(); for (const r of rows) { read++; const id = String(r[iId] || '').trim(); if (id) { if (seen.has(id)) { dupes++; continue; } seen.add(id); } const ts = Number(r[iSt]); if (!ts) { noTs++; continue; } const loc = ukLocal(ts); const dir = String(r[iDir] || '').trim(); const user = String(r[iUser] || '').trim(); const answered = r[iAns] !== '' && r[iAns] !== null; const dur = Number(r[iDur] || 0); const reason = String(r[iReason] || '').trim(); let talk = 0; if (answered && r[iEnd] !== '' && r[iEnd] !== null) { const s = Number(r[iEnd]) - Number(r[iAns]); if (s > 0) talk = s; } let team = 'Other'; if (dir === 'inbound') { const tn = String(r[iTeam] || '').trim(); const num = normNum(r[iNum]); let how = 'unattributed'; if (tn) { team = tn; how = 'team_name'; } else if (numberTeam[num]) { team = numberTeam[num]; how = 'cfg_numbers'; } else if (NUMBER_TEAM[num]) { team = NUMBER_TEAM[num]; how = 'built_in_list'; } attrib[how]++; const it = inboundTeams[team] || (inboundTeams[team] = { calls: 0, answered: 0, viaNumber: 0 }); it.calls++; if (answered) it.answered++; if (how === 'cfg_numbers' || how === 'built_in_list') it.viaNumber++; } else if (dir === 'outbound') { team = agentTeam[user] || 'Unmapped'; } weekCounts[loc.weekKey] = (weekCounts[loc.weekKey] || 0) + 1; // Hourly team aggregate const hk = loc.date + '|' + loc.hour + '|' + dir + '|' + team; let h = hourly[hk]; if (!h) { h = { weekKey: loc.weekKey, isoYear: loc.isoYear, isoWeek: loc.isoWeek, date: loc.date, weekday: loc.weekday, hour: loc.hour, direction: dir, team: team, calls: 0, answered: 0, missed: 0, missedOoh: 0, missedAbandoned: 0, connected10s: 0, talkSecs: 0 }; hourly[hk] = h; } h.calls++; if (answered) h.answered++; else { h.missed++; if (reason === 'out_of_opening_hours') h.missedOoh++; else if (reason.indexOf('abandoned') >= 0) h.missedAbandoned++; } if (dir === 'outbound' && answered && dur >= 10) h.connected10s++; h.talkSecs += talk; // Agent weekly aggregate if (user) { const ak = loc.weekKey + '|' + user; let a = agents[ak]; if (!a) { a = { weekKey: loc.weekKey, isoYear: loc.isoYear, isoWeek: loc.isoWeek, user: user, team: agentTeam[user] || 'Unmapped', inboundAnswered: 0, outboundTotal: 0, outboundConnected10s: 0, talkInSecs: 0, talkOutSecs: 0 }; agents[ak] = a; } if (dir === 'inbound') { if (answered) a.inboundAnswered++; a.talkInSecs += talk; } if (dir === 'outbound') { a.outboundTotal++; if (answered && dur >= 10) a.outboundConnected10s++; a.talkOutSecs += talk; } } } } const hourlyOut: (string | number)[][] = [['week_key', 'iso_year', 'iso_week', 'date', 'weekday', 'hour', 'direction', 'team', 'calls', 'answered', 'missed', 'missed_ooh', 'missed_abandoned', 'connected_10s', 'talk_secs']]; Object.keys(hourly).sort().forEach(k => { const h = hourly[k]; hourlyOut.push([h.weekKey, h.isoYear, h.isoWeek, h.date, h.weekday, h.hour, h.direction, h.team, h.calls, h.answered, h.missed, h.missedOoh, h.missedAbandoned, h.connected10s, h.talkSecs]); }); const agentOut: (string | number)[][] = [['week_key', 'iso_year', 'iso_week', 'user_name', 'team', 'inbound_answered', 'outbound_total', 'outbound_connected_10s', 'talk_in_secs', 'talk_out_secs']]; Object.keys(agents).sort().forEach(k => { const a = agents[k]; agentOut.push([a.weekKey, a.isoYear, a.isoWeek, a.user, a.team, a.inboundAnswered, a.outboundTotal, a.outboundConnected10s, a.talkInSecs, a.talkOutSecs]); }); writeSheet(workbook, 'agg_calls_hourly', hourlyOut, [3]); writeSheet(workbook, 'agg_agent_week', agentOut, []); console.log('raw rows read: ' + read + ' | duplicate ids skipped: ' + dupes + ' | rows without started_at: ' + noTs); console.log('agg_calls_hourly rows: ' + (hourlyOut.length - 1) + ' | agg_agent_week rows: ' + (agentOut.length - 1)); console.log('Inbound attribution: ' + Object.keys(attrib).map(k => k + ' ' + attrib[k]).join(' | ')); console.log('Inbound teams (all weeks) — team: calls / answered / of which attributed by number'); Object.keys(inboundTeams).sort((a, b) => inboundTeams[b].calls - inboundTeams[a].calls).forEach(t => { const x = inboundTeams[t]; console.log(' ' + t + ': ' + x.calls + ' / ' + x.answered + ' / ' + x.viaNumber); }); console.log('Calls per week (UK time):'); Object.keys(weekCounts).sort().forEach(k => console.log(' ' + k + ': ' + weekCounts[k])); } // Two-column lookup from a named Table (preferred) or a sheet of the same name (with or without trailing space). function readLookup(workbook: ExcelScript.Workbook, name: string, keyCol: string, valCol: string, numericKey: boolean): { [k: string]: string } { let values: (string | number | boolean)[][] = []; const tbl = workbook.getTable(name); if (tbl) values = tbl.getRange().getValues(); else { const ws = workbook.getWorksheet(name) || workbook.getWorksheet(name + ' '); if (ws) values = ws.getUsedRange().getValues(); } const out: { [k: string]: string } = {}; if (values.length < 2) { console.log('WARNING: ' + name + ' not found or empty'); return out; } const h = values[0].map(x => String(x).trim()); const ki = h.indexOf(keyCol), vi = h.indexOf(valCol); if (ki < 0 || vi < 0) { console.log('WARNING: ' + name + ' is missing column ' + (ki < 0 ? keyCol : valCol)); return out; } for (let i = 1; i < values.length; i++) { const k = numericKey ? normNum(values[i][ki]) : String(values[i][ki] || '').trim(); const v = String(values[i][vi] || '').trim(); if (k && v) out[k] = v; } return out; } // Number IDs can arrive as 202266, 202266.0 or '202266 ' — normalise to '202266'. function normNum(v: string | number | boolean): string { const s = String(v === null || v === undefined ? '' : v).trim(); if (!s) return ''; const n = Number(s); return isNaN(n) ? s : String(Math.round(n)); } // Clear (or create) a sheet and write rows in chunks. textCols = column indexes to keep as text (e.g. dates). function writeSheet(workbook: ExcelScript.Workbook, name: string, rows: (string | number)[][], textCols: number[]) { const ws: ExcelScript.Worksheet = workbook.getWorksheet(name) || workbook.addWorksheet(name); ws.getRange().clear(ExcelScript.ClearApplyTo.contents); const cols = rows[0].length; textCols.forEach(c => ws.getRangeByIndexes(0, c, Math.max(rows.length, 1), 1).setNumberFormatLocal('@')); for (let s = 0; s < rows.length; s += WRITE_CHUNK) { const part = rows.slice(s, s + WRITE_CHUNK); ws.getRangeByIndexes(s, 0, part.length, cols).setValues(part); } } // Convert a Unix timestamp to UK local time (GMT/BST) and its ISO week. function ukLocal(ts: number): { date: string; hour: number; weekday: number; isoYear: number; isoWeek: number; weekKey: string } { const ms = ts * 1000; const y = new Date(ms).getUTCFullYear(); const bstStart = lastSundayUtc1am(y, 2); const bstEnd = lastSundayUtc1am(y, 9); const offset = (ms >= bstStart && ms < bstEnd) ? 3600000 : 0; const d = new Date(ms + offset); // read with UTC getters = UK wall-clock time const date = d.getUTCFullYear() + '-' + pad(d.getUTCMonth() + 1) + '-' + pad(d.getUTCDate()); const weekday = d.getUTCDay() || 7; // 1 = Mon ... 7 = Sun const t = new Date(Date.UTC(d.getUTCFullYear(), d.getUTCMonth(), d.getUTCDate() + 4 - weekday)); const isoYear = t.getUTCFullYear(); const isoWeek = Math.ceil(((t.getTime() - Date.UTC(isoYear, 0, 1)) / 86400000 + 1) / 7); return { date, hour: d.getUTCHours(), weekday, isoYear, isoWeek, weekKey: isoYear + '-W' + pad(isoWeek) }; } // Clocks change at 01:00 UTC on the last Sunday of March (month 2) and October (month 9). function lastSundayUtc1am(year: number, month: number): number { const last = new Date(Date.UTC(year, month + 1, 0)); return Date.UTC(year, month, last.getUTCDate() - last.getUTCDay(), 1); } function pad(n: number): string { return (n < 10 ? '0' : '') + n; }
